export class AudioPlayer {
  constructor(audioUrl, mediaId) {
    this.audio = new Audio(audioUrl);
    this.mediaId = mediaId;
  }

  savePlaybackProgress() {
    const elapsed = this.audio.currentTime;
    localStorage.setItem(`playback_progress_${this.mediaId}`, elapsed);
  }

  loadPlaybackProgress() {
    return parseFloat(localStorage.getItem(`playback_progress_${this.mediaId}`)) || 0;
  }
}

import { useState, useEffect } from 'react';

export const useChallengeProgress = (totalDays = 7) => {
  const [completedDays, setCompletedDays] = useState<number[]>([]);

  const toggleDayComplete = (day: number) => {
    setCompletedDays(prev => 
      prev.includes(day) ? prev.filter(d => d !== day) : [...prev, day]
    );
  };

  const isFullyComplete = completedDays.length === totalDays;
  return { completedDays, toggleDayComplete, isFullyComplete };
};
